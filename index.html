Script Tự Động Hút Tin Nhắn Từ Zalo Desktop (Windows)

Vì ứng dụng Zalo Desktop trên Windows đóng API và không cho xuất file trực tiếp, bạn có thể lưu đoạn mã Python dưới đây thành file zalo_export.py (cài thư viện pip install uiautomation) để tự động cuộn và xuất tin nhắn nhóm đang mở ra file .txt nạp vào web:

# Script tự động đọc khung chat Zalo Desktop đang mở trên Windows
# Cài đặt 1 lần: pip install uiautomation
import uiautomation as auto
import time, datetime

zalo_win = auto.WindowControl(searchDepth=1, RegexName="Zalo.*")
if not zalo_win.Exists(3):
    print("Không tìm thấy cửa sổ Zalo Desktop!")
    exit()

zalo_win.SetActive()
lines = []
print("Đang quét tin nhắn... Bấm Ctrl+C để dừng và lưu file.")
for _ in range(15):  # Số lần cuộn lên lịch sử cũ (tăng lên nếu cần nhiều ngày)
    for ctrl, _ in auto.WalkControl(zalo_win, maxDepth=12):
        if ctrl.ControlTypeName == "TextControl" and len(ctrl.Name.strip()) > 3:
            txt = ctrl.Name.strip()
            if txt not in lines:
                lines.append(txt)
    zalo_win.WheelUp(wheelTimes=8)
    time.sleep(0.6)

fname = f"Zalo_Export_{datetime.datetime.now().strftime('%Y%m%d_%H%M')}.txt"
with open(fname, "w", encoding="utf-8") as f:
    f.write("\n".join(reversed(lines)))
print(f"Đã xuất {len(lines)} dòng ra file: {fname}")
